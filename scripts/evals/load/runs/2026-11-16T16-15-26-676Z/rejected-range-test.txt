import assert from "node:assert/strict";
import { randomBytes } from "node:crypto";
import { test } from "node:test";

import { openRangeFile } from "../src/range-file";

function serveFile(data: Uint8Array, ranges = true) {
  let bytes = 0;
  const server = Bun.serve({
    port: 0,
    fetch(req) {
      const m = /bytes=(\d+)-(\d+)/.exec(req.headers.get("range") ?? "");
      if (ranges && m) {
        const start = Number(m[1]), end = Math.min(Number(m[2]) + 1, data.length);
        const body = data.slice(start, end);
        bytes += body.length;
        return new Response(body, {
          status: 206,
          headers: { "content-range": `bytes ${start}-${end - 1}/${data.length}`, etag: '"file-1"' },
        });
      }
      bytes += data.length;
      return new Response(data);
    },
  });
  return { url: String(server.url), bytes: () => bytes, close: () => server.stop(true) };
}

test("remote slices match file bytes across block boundaries without downloading the unused file", async (t) => {
  const data = randomBytes(2 * 1024 * 1024), server = serveFile(data);
  t.after(server.close);
  const file = await openRangeFile(server.url);
  assert.equal(file.size, data.length);
  assert.deepEqual(await file.slice(65530, 65550), new Uint8Array(data.subarray(65530, 65550)));
  assert.deepEqual(await file.slice(123, 789), new Uint8Array(data.subarray(123, 789)));
  assert.deepEqual(await file.slice(data.length - 7, data.length), new Uint8Array(data.subarray(-7)));
  assert.ok(server.bytes() < data.length / 2, `read ${server.bytes()} of ${data.length} bytes`);
  const first = server.bytes();
  await file.slice(65530, 65550);
  assert.equal(server.bytes(), first);
});

test("a server without ranges supplies every slice from its one full response", async (t) => {
  const data = randomBytes(77777), server = serveFile(data, false);
  t.after(server.close);
  const file = await openRangeFile(server.url);
  assert.deepEqual(await file.slice(0, 123), new Uint8Array(data.subarray(0, 123)));
  assert.deepEqual(await file.slice(123, data.length), new Uint8Array(data.subarray(123)));
  assert.equal(server.bytes(), data.length);
});

test("HTTP failures remain errors rather than an empty file", async (t) => {
  const server = Bun.serve({ port: 0, fetch: () => new Response(null, { status: 404 }) });
  t.after(() => server.stop(true));
  await assert.rejects(openRangeFile(String(server.url)), /404/);
});
