import assert from "node:assert/strict";
import { mkdtempSync, readFileSync, rmSync, statSync, unlinkSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { test } from "node:test";
import { brotliDecompressSync } from "node:zlib";
import { strToU8, unzipSync, zipSync } from "fflate";

import { slimJar } from "../mc/slim_jar.mjs";

test("the slim archive's compressed directory index retains every asset and its original ZIP offsets", (t) => {
  const dir = mkdtempSync(join(tmpdir(), "studio-jar-index-"));
  t.after(() => rmSync(dir, { recursive: true, force: true }));
  const files = { "assets/test/a.json": strToU8('{"test":true}'), "data/test/a.json": strToU8("{}") };
  writeFileSync(join(dir, "client.jar"), zipSync(files));
  slimJar({ mcDir: dir });
  const archive = readFileSync(join(dir, "client_assets.zip"));
  const indexBytes = readFileSync(join(dir, "client_assets.zip.index.json"));
  const index = JSON.parse(indexBytes);
  assert.equal(index.size, archive.length);
  assert.equal(index.tag, `${archive.length}-${Math.round(statSync(join(dir, "client_assets.zip")).mtimeMs)}`);
  const directory = Buffer.from(index.directory, "base64");
  assert.deepEqual(directory, archive.subarray(index.offset));
  assert.equal(directory.readUInt32LE(0), 0x02014b50);
  assert.deepEqual(unzipSync(archive), files);
  assert.deepEqual(brotliDecompressSync(readFileSync(join(dir, "client_assets.zip.index.json.br"))), indexBytes);
});

test("an existing slim archive acquires a missing index without rewriting the archive", (t) => {
  const dir = mkdtempSync(join(tmpdir(), "studio-jar-index-"));
  t.after(() => rmSync(dir, { recursive: true, force: true }));
  writeFileSync(join(dir, "client.jar"), zipSync({ "assets/test/a.json": strToU8("{}") }));
  slimJar({ mcDir: dir });
  const archive = join(dir, "client_assets.zip"), before = statSync(archive).mtimeMs;
  unlinkSync(`${archive}.index.json`);
  slimJar({ mcDir: dir });
  assert.equal(statSync(archive).mtimeMs, before);
  assert.equal(JSON.parse(readFileSync(`${archive}.index.json`)).size, statSync(archive).size);
});
