import assert from "node:assert/strict";
import { test } from "node:test";

import { withSoundLevel } from "../app/sound/level";

test("a decoded buffer can play without first reading its PCM for loudness matching", () => {
  const buf = {
    numberOfChannels: 1,
    getChannelData: () => {
      throw new Error("PCM was scanned");
    },
  };
  const sound = withSoundLevel(buf);
  assert.equal(sound.buf, buf);
  assert.throws(() => sound.mean, /PCM was scanned/);
});

test("loudness matching measures the energy of every channel", () => {
  const channels = [new Float32Array([0.5, -0.5]), new Float32Array([1, -1])];
  const sound = withSoundLevel({
    numberOfChannels: 2,
    getChannelData: (i: number) => channels[i]!,
  });
  assert.ok(Math.abs(sound.mean - -2.041199826559248) < 1e-12);
});

test("silent sounds retain the existing finite loudness floor", () => {
  const sound = withSoundLevel({ numberOfChannels: 1, getChannelData: () => new Float32Array(12) });
  assert.equal(sound.mean, -120);
});
